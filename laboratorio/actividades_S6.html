import { test, expect, Page } from '@playwright/test';

const BASE_URL = 'http://127.0.0.1:4000';
const USUARIO = 'ana';
const CLAVE = 'Clave123';

// Lee el último código de seguridad (6 dígitos) del buzón de SMS de pruebas.
async function leerCodigoDelBuzon(page: Page, usuario: string): Promise<string> {
  const buzon = await page.context().newPage();
  await buzon.goto(`${BASE_URL}/buzon`);
  await buzon.getByRole('textbox').fill(usuario);
  await buzon.getByRole('button', { name: 'Buscar' }).click();
  const texto = await buzon.locator('body').innerText();
  const codigos = texto.match(/\b\d{4,8}\b/g);
  await buzon.close();
  if (!codigos || codigos.length === 0) {
    throw new Error('No se encontró un código de seguridad en el buzón');
  }
  return codigos[codigos.length - 1];
}

async function iniciarSesion(page: Page) {
  await page.goto(BASE_URL);
  await page.getByLabel(/usuario/i).fill(USUARIO);
  await page.getByLabel(/clave/i).fill(CLAVE);
  await page.getByRole('button', { name: /ingresar|iniciar sesión/i }).click();

  await expect(page.getByLabel('Código de seguridad')).toBeVisible();
  const codigo = await leerCodigoDelBuzon(page, USUARIO);
  await page.getByLabel('Código de seguridad').fill(codigo);
  await page.getByRole('button', { name: 'Validar' }).click();
}

test.describe('Recargas', () => {
  test('CP-102 · Recarga de más de $50.000 con código', async ({ page }) => {
    // Precondición: Ana inició sesión
    await iniciarSesion(page);

    // Paso 1
    await page.goto(`${BASE_URL}/recargas`);

    // Paso 2
    await page.getByLabel(/operador/i).selectOption({ label: 'Claro' });
    await page.getByLabel(/celular/i).fill('3001234567');
    await page.getByLabel(/valor/i).fill('60000');

    // Paso 3
    await page.getByRole('button', { name: 'Recargar' }).click();

    // Esperado 1: antes de recargar, pide el código de seguridad
    const campoCodigo = page.getByLabel('Código de seguridad');
    await expect(campoCodigo).toBeVisible();

    // Paso 4: nuevo código desde el buzón
    const codigo = await leerCodigoDelBuzon(page, USUARIO);
    await campoCodigo.fill(codigo);
    await page.getByRole('button', { name: 'Confirmar recarga' }).click();

    // Esperado 2: comprobante
    await expect(page.getByText('Recarga exitosa')).toBeVisible();
    await expect(page.getByText('Valor: $60.000')).toBeVisible();

    // Esperado 3: saldo de Ahorros = $1.440.000
    await page.goto(BASE_URL + '/');
    await expect(page.getByText('$1.440.000')).toBeVisible();
  });
});

Ver el código, tal cual
import { test, expect } from '@playwright/test';

const BASE = 'http://127.0.0.1:4000';
const ORIGEN = '1000000002';  // Corriente
const DESTINO = '2000000001';
const MONTO = 100000;

test('Transferencia de $100.000 desde Corriente: respuesta y saldo final', async ({ request }) => {
  // Autenticación: login -> SMS en buzón -> OTP -> token
  const login = await request.post(`${BASE}/api/login`, { data: { usuario: 'ana', clave: 'Clave123' } });
  expect(login.ok()).toBeTruthy();
  const { idDesafio } = await login.json();

  const buzon = await (await request.get(`${BASE}/api/buzon?usuario=ana`)).json();
  const codigo = buzon[0].codigo;

  const otp = await request.post(`${BASE}/api/otp`, { data: { idDesafio, codigo } });
  expect(otp.ok()).toBeTruthy();
  const { token } = await otp.json();
  const headers = { Authorization: `Bearer ${token}` };

  const saldoCorriente = async () => {
    const cuentas = await (await request.get(`${BASE}/api/cuentas`, { headers })).json();
    return cuentas.find((c: any) => c.numero === ORIGEN).saldo as number;
  };

  const saldoInicial = await saldoCorriente();
  expect(saldoInicial).toBeGreaterThanOrEqual(MONTO + 400);

  const res = await request.post(`${BASE}/api/transferencias`, {
    headers,
    data: { origen: ORIGEN, destino: DESTINO, monto: MONTO },
  });
  expect(res.status()).toBe(201);
  const body = await res.json();

  // Respuesta de la transferencia
  expect(typeof body.comprobante).toBe('number');
  expect(body.monto).toBe(MONTO);
  expect(body.gmf).toBe(400);                 // 4 x 1000 sobre 100.000
  expect(body.total).toBe(MONTO + body.gmf);  // 100.400

  // Saldo final: lo debitado debe ser el total (monto + GMF)
  const saldoFinal = await saldoCorriente();
  expect(body.saldo).toBe(saldoFinal);
  expect(saldoFinal).toBe(saldoInicial - body.total);
});

El boleto de salida de la S5. En el chat, las cuatro letras en orden, todos al mismo tiempo. Por ejemplo: «a c b a».

  1. El botón «Recargar» pasa a llamarse «Recargar ya». ¿Dónde se corrige?

    a) en cada prueba que recarga · b) en PaginaRecargas · c) en la fábrica

  2. Dos pruebas usan «el usuario de siempre» y una falla porque la otra le cambió el saldo. ¿Qué lo evita?

    a) un Singleton del usuario · b) otro Page Object · c) una fábrica que entrega un cliente nuevo

  3. Un cliente crea una transferencia sobre el tope y un aprobador la autoriza. ¿Qué patrón cuenta mejor esa prueba?

    a) Screenplay · b) Singleton · c) Factory

  4. ¿Qué debe existir una sola vez en toda la corrida?

    a) el navegador · b) la dirección del ambiente · c) la sesión del cliente

Y una más, en una palabra: Factory, ¿es un patrón de pruebas o creacional?

  1. b · Page Object: el localizador vive en un solo lugar.
  2. c · Factory: cada llamada entrega un objeto nuevo, y ...cambios solo pisa lo que la prueba pidió («el combo 2, sin cebolla»). Si los cambios se quedaran, la prueba siguiente recibiría datos de la anterior.
  3. a · Screenplay: dos actores con habilidades distintas.
  4. b · Singleton solo para la configuración; navegador y sesión, uno por prueba.

Factory es creacional, igual que Singleton y Builder: vienen del libro de Gamma, de desarrollo. En QA los usamos para los datos y la configuración. Page Object y Screenplay sí son patrones de automatización de pruebas. Page Object no es «el viejo»: es el más usado y el que recomienda la documentación de Playwright. Screenplay no lo reemplaza: una tarea usa páginas por dentro. Un framework combina varios patrones, uno por problema.

ficha · 3 mincapa: src/config/

Configuración y ambientes

En la S3, entorno.ts ya era un solo lugar para la dirección del banco. En la S5 le pusimos nombre: Singleton. Hoy le ponemos los controles que exige un banco.

El problema que ya sufrías a mano
El caso pasa en QA y falla en UAT «porque allá la clave es otra». Alguien corre la regresión contra el ambiente equivocado. Y la clave de prueba termina escrita en un Excel, en un correo o en la evidencia.
La idea, en una frase
Un ambiente es un control, no solo una URL. La prueba es la misma en QA y en UAT. Lo que cambia (dirección, usuario, clave y datos) vive en un solo lugar, congelado y sin secretos en el código.
Sus partes, en el repo
src/config/ambientes.ts: el catálogo local | qa | uat y configurarAmbiente(nombre, variables) · src/config/entorno.ts: el resultado, congelado con Object.freeze · las variables de la terminal: $env:AMBIENTE = 'qa' y $env:CLAVE_BANCO = '...' · .env.ejemplo: los nombres de las variables, sin valores.
Sí, cuando
Todo lo que cambia entre QA, UAT y preproducción: dirección, usuarios, claves, tiempos y los datos de cada ambiente. Es el environment de Postman, pero para todo el framework.
No, cuando
Una clave escrita en el código o en un valor compartido de Postman, que se sincroniza con la nube. Un objeto de configuración que una prueba puede modificar, porque daña a la siguiente. Y nunca un ambiente produccion en el catálogo.
La frase para tu líder
«Las pruebas no pueden correr contra producción por error. Si falta una clave, se nota en el primer segundo. Y ningún secreto queda en el repo.»
Cómo pedírselo a la IA
«La dirección y las claves salen de entorno. No las escribas en la prueba ni inventes una URL.»

demo · 2 min · pasa sola, falla en la corrida

¿Por qué se congela?

test('A · cambia el usuario «de afán»', () => {
  entorno.usuario = 'luis';
});

test('B · espera el de siempre', () => {
  expect(entorno.usuario).toBe('ana');
});
✘ B · espera el de siempre

  Expected: "ana"
  Received: "luis"

Corrida sola, la B pasa. En la corrida completa falla, porque la A le cambió el usuario. Es «la prueba que a veces falla» que todos conocen.

Con Object.freeze, el editor la frena antes de correr: Cannot assign to 'usuario' because it is a read-only property. Y si alguien la corre igual: TypeError: Cannot assign to read only property 'usuario'.

Lo que se queda entre pruebas es un objeto compartido, como entorno: por eso se congela. La fábrica es lo contrario: cada llamada entrega un objeto nuevo, y ...cambios solo pisa lo que esa prueba pidió.

¿Y usuarios.ana no debería generarse? Depende de quién más usa el dato. En el Banco Demo, Ana existe siempre y sus saldos se reinician: es un dato con nombre, en un solo lugar, que se cambia por ambiente (USUARIO_BANCO). En un sitio compartido como ParaBank sí se genera (fabricaDeClientes), porque otro lo podría tocar.

ficha · 3 mincapa: src/paginas/

Localizadores robustos

En la S5 los localizadores se fueron al Page Object: son lo que la pantalla tiene. Hoy vemos cuál elegir para que sobreviva a un rediseño.

El problema que ya sufrías a mano
Desarrollo «solo cambió el diseño» y se cae la regresión entera, aunque para el cliente la pantalla hace exactamente lo mismo.
La idea, en una frase
Busca cada elemento como lo busca una persona: por su rol y su nombre, por la etiqueta del campo o por su texto. Si no hay nada visible y estable, usa un data-testid pactado con desarrollo. CSS y XPath, nunca.
Sus partes, en el repo
getByRole('button', { name: 'Recargar' }) · getByLabel('Número de celular') · getByText('Recarga exitosa') · getByTestId('saldo-disponible'). Todos son private dentro del Page Object, en src/paginas/.
Sí, cuando
Siempre, empezando por el peldaño más alto de la escalera. Además pruebas accesibilidad: si getByLabel no encuentra el campo, un lector de pantalla tampoco.
No, cuando
Clases CSS (cambian con el diseño), nth-child (cambia con el orden), XPath absoluto (cambia con cualquier contenedor) e ids que cambian con cada versión. Tampoco un localizador dentro de la prueba: vive en el Page Object.
La frase para tu líder
«Si rediseñan la pantalla y el negocio no cambió, la regresión sigue en verde. Si se cae, es porque algo cambió para el cliente.»
Cómo pedírselo a la IA

ficha · 3 mincapa: src/paginas/ y tests/

Esperas

La pregunta del jueves: «¿el tiempo de espera se puede programar?». Sí se puede, pero casi nunca hace falta.

El problema que ya sufrías a mano
En manual uno espera «a que cargue» mirando la pantalla. En la prueba alguien escribe «espera 10 segundos». Si el sistema tarda 11, falla. Si tarda 1, se pierden 9 en cada prueba, todas las noches.
La idea, en una frase
Playwright ya espera solo. Cada acción espera a que el elemento esté visible y habilitado. Cada await expect(...) reintenta hasta que se cumpla (5 segundos por defecto). Tu trabajo es poner await en cada paso.
Sus partes, en el repo
await: cada acción del navegador devuelve una promesa («ya te respondo»), y await quiere decir «espera la respuesta antes de seguir» · las acciones (click, fill) esperan solas · las aserciones que reintentan: await expect(locator).toHaveText(...) · los tiempos: 5 s por expect y 30 s por prueba (Test timeout of 30000ms exceeded).
Sí, cuando
Si un sistema es lento de verdad, el tiempo se sube en un solo lugar (expect: { timeout } en playwright.config.ts) o solo en esa aserción ({ timeout: 10_000 }). Y la lentitud se reporta como hallazgo.
No, cuando
page.waitForTimeout(10000): hace la prueba lenta e inestable a la vez. Tampoco expect(await locator.innerText()), que lee una sola vez y no reintenta. Ni un paso sin await: a veces pasa por suerte, y eso es peor.
La frase para tu líder
«Las pruebas esperan lo justo: no fallan porque el sistema tardó un segundo más, ni pierden minutos esperando por si acaso.»
Cómo pedírselo a la IA
«Sin waitForTimeout. Pon await en cada paso y usa await expect(locator) con aserciones que reintentan.»

¿Cuánto espera Playwright? Cada expect, 5 segundos. Cada prueba completa, 30 (Test timeout of 30000ms exceeded). Si 4 pruebas se quedan esperando algo que nunca llega, son 4 × 30 segundos: los «2 minutos» de una corrida que se cuelga.

Los datos, fijos y con nombre. valida() siempre entrega Claro, 3001234567 y $10.000, porque una prueba que falla tiene que poder repetirse igual. Aleatorio solo cuando sirve (el cliente único de ParaBank) y siempre reproducible. Si una regla cambia, se cambia el valor en un solo lugar: la fábrica.
